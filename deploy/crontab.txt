# TradeAlert crontab per VM Oracle Free Tier (Ubuntu 22.04).
#
# Tutti gli orari sono in TZ Europe/Rome (impostata sulla VM).
# Quiet hours interno blocca le notifiche fuori dalle finestre attive
# (lun-ven 07:00-22:30, weekend 09:00-22:30).
#
# Installazione:
#   sudo timedatectl set-timezone Europe/Rome
#   crontab deploy/crontab.txt
#   crontab -l   # verifica
#
# Variabili: lo script load_config() legge da /home/ubuntu/tradealert/.env

SHELL=/bin/bash
PATH=/home/ubuntu/tradealert/.venv/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
TRADEALERT_HOME=/home/ubuntu/tradealert

# ---------- Market Scan: ogni 60 min in finestra attiva (2026-04-24: ridotto da 30') ----------
5 7-22 * * *         cd $TRADEALERT_HOME && python -m jobs.morning_scan >> logs/scan.log 2>&1

# ---------- Position Monitor: ogni 30 min, 7-22 IT (con LLM per close) ----------
*/30 7-22 * * *      cd $TRADEALERT_HOME && python -m jobs.monitor >> logs/monitor.log 2>&1

# ---------- Trailing stop leggero: ogni 5 min H24 (no LLM, solo migliorativo) ----------
*/5 * * * *          cd $TRADEALERT_HOME && python -m jobs.trailing_stop >> logs/trailing.log 2>&1

# ---------- Logging high/low intra-trade: ogni 30 min H24 (Sprint 3, read+append) ----------
*/30 * * * *         cd $TRADEALERT_HOME && python -m jobs.intra_trade_log >> logs/intra_trade.log 2>&1

# ---------- Shadow check scoring: 1x/giorno 22:30, report Telegram a campione pronto (Sprint 4 t1) ----------
30 22 * * *          cd $TRADEALERT_HOME && python -m jobs.shadow_check >> logs/shadow_check.log 2>&1

# ---------- Briefing: DISABILITATI (2026-04-24) per taglio costi API (~€6/mese).
# Doppio gate: crontab commentato qui + guard BRIEFING_ENABLED in jobs/briefing.py.
# Per riabilitare: scommenta le 3 righe sotto, imposta BRIEFING_ENABLED=true nel
# .env della VM e reinstalla il crontab (crontab deploy/crontab.txt).
# 2 8 * * *            cd $TRADEALERT_HOME && python -m jobs.briefing morning   >> logs/briefing.log 2>&1
# 2 16 * * *           cd $TRADEALERT_HOME && python -m jobs.briefing afternoon >> logs/briefing.log 2>&1
# 2 22 * * *           cd $TRADEALERT_HOME && python -m jobs.briefing evening   >> logs/briefing.log 2>&1

# ---------- Health VM Oracle: 1x/giorno (banda, disco, RAM, listener) ----------
10 9 * * *           cd $TRADEALERT_HOME && python -m jobs.health_check >> logs/health.log 2>&1

# ---------- Weekly report: domenica 20:00 Europe/Rome ----------
0 20 * * 0           cd $TRADEALERT_HOME && python -m jobs.weekly_report >> logs/weekly.log 2>&1

# ---------- Macro scan eventi critici dalle news: 06:50 prima del primo scanner ----------
50 6 * * *           cd $TRADEALERT_HOME && python -m jobs.macro_scan >> logs/macro.log 2>&1

# ---------- Reconcile DB vs Capital: ogni ora (housekeeping) ----------
15 * * * *           cd $TRADEALERT_HOME && python -m jobs.reconcile >> logs/reconcile.log 2>&1

# Il listener Telegram NON va in crontab: gira come daemon systemd
# (vedi tradealert-listener.service) per latenza ~secondi sui comandi.
